function groupedHTML(){
  const active=state.books.filter(b=>!b.trashed);
  let groups=[];
  if(state.view==="authors"){
    const m=groupBy(active,b=>b.author||"Unknown author");
    groups=[...m].map(([k,v])=>({name:k,subtitle:`${v.length} books`,books:v}));
  } else if(state.view==="series"){
    const m=groupBy(active.filter(b=>b.series),b=>b.series);
    groups=[...m].map(([k,v])=>({name:k,subtitle:`${v.length} titles`,books:v}));
  } else if(state.view==="collections"){
    const m=new Map();
    active.forEach(b=>(b.collections||[]).forEach(c=>{ if(!m.has(c))m.set(c,[]);m.get(c).push(b); }));
    groups=[...m].map(([k,v])=>({name:k,subtitle:`${v.length} items`,books:v}));
  } else if(state.view==="formats"){
    const m=groupBy(active,b=>b.format||"Unknown");
    groups=[...m].map(([k,v])=>({name:k,subtitle:`${v.length} files`,books:v}));
  } else if(state.view==="downloads"){
    groups=[{name:"Imported files",subtitle:`${active.filter(b=>b.imported).length} items`,books:active.filter(b=>b.imported)}];
  } else {
    groups=[{name:"Browser library",subtitle:"Files are stored locally in IndexedDB",books:active}];
  }
  groups.sort((a,b)=>a.name.localeCompare(b.name));
  return `<div class="section-head"><div><h2>${viewTitle()}</h2><p>Organize and browse your library</p></div></div>
  <div class="book-list">${groups.length?groups.map(g=>`
    <article class="book-row clickable" style="min-height:96px;grid-template-columns:58px minmax(0,1fr) auto" data-group="${escapeHTML(g.name)}">
      <div style="width:58px;height:70px;border-radius:9px;background:#dceced;color:#136c78;display:grid;place-items:center">${I(state.view==="authors"?"person":state.view==="formats"?"stack":"books")}</div>
      <div class="book-main"><div class="book-title">${escapeHTML(g.name)}</div><div class="book-author">${escapeHTML(g.subtitle)}</div><div class="book-meta">${g.books.slice(0,3).map(b=>escapeHTML(b.title)).join(" · ")}</div></div>
      <div class="book-actions">${I("chevron")}</div>
    </article>`).join(""):'<div class="empty">No groups yet.</div>'}</div>`;
}
function groupBy(arr,fn){const m=new Map();for(const x of arr){const k=fn(x);if(!m.has(k))m.set(k,[]);m.get(k).push(x)}return m;}
